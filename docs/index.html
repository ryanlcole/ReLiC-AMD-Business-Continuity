3. Affected-record resolver PROOF OF WORK

ReLiC intersects scope and valid time, then returns explanations instead of updates.

def reveal_affected(order, records):
    affected = []
    for record in records:
        in_time = record.event_date >= order.backpay_from
        in_scope = record.classification in order.classifications
        if in_time and in_scope:
            affected.append({
                "record_id": record.id,
                "authority": order.id,
                "reason": "event date + classification intersect authority",
                "action": "REVIEW_REQUIRED"
            })
    return affected

# There is intentionally NO update(), delete(), pay(), or write-back here.
Boundary: ReLiC may reveal 14,283 affected records. That does not mean 14,283 records have been changed.
